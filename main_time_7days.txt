#include "raylib.h"
#include "raymath.h"
#include <cmath>

static float ClampFloat(float value, float minValue, float maxValue)
{
    if (value < minValue) return minValue;
    if (value > maxValue) return maxValue;
    return value;
}

static Color LerpColor(Color a, Color b, float amount)
{
    amount = ClampFloat(amount, 0.0f, 1.0f);

    Color result{};
    result.r = (unsigned char)ClampFloat((float)a.r + ((float)b.r - (float)a.r) * amount, 0.0f, 255.0f);
    result.g = (unsigned char)ClampFloat((float)a.g + ((float)b.g - (float)a.g) * amount, 0.0f, 255.0f);
    result.b = (unsigned char)ClampFloat((float)a.b + ((float)b.b - (float)a.b) * amount, 0.0f, 255.0f);
    result.a = 255;
    return result;
}

struct Note
{
    Vector3 position;
    const char* title;
    const char* text;
    bool found;
};

struct CollisionBox
{
    Vector3 position;
    Vector3 size;
};

static bool CheckPlayerCollision(Vector3 playerPosition, float playerRadius, const CollisionBox& box)
{
    float minX = box.position.x - box.size.x / 2.0f;
    float maxX = box.position.x + box.size.x / 2.0f;
    float minZ = box.position.z - box.size.z / 2.0f;
    float maxZ = box.position.z + box.size.z / 2.0f;

    float closestX = ClampFloat(playerPosition.x, minX, maxX);
    float closestZ = ClampFloat(playerPosition.z, minZ, maxZ);

    float dx = playerPosition.x - closestX;
    float dz = playerPosition.z - closestZ;

    return (dx * dx + dz * dz) < (playerRadius * playerRadius);
}

static bool IsNearAndLookingAt(Camera3D camera, Vector3 objectPosition, float distanceLimit, float dotLimit)
{
    Vector3 toObject = Vector3Subtract(objectPosition, camera.position);
    float distance = Vector3Length(toObject);

    if (distance > distanceLimit || distance < 0.001f)
        return false;

    toObject = Vector3Normalize(toObject);

    Vector3 direction = Vector3Subtract(camera.target, camera.position);

    if (Vector3Length(direction) < 0.001f)
        return false;

    direction = Vector3Normalize(direction);

    return Vector3DotProduct(direction, toObject) > dotLimit;
}


enum class TimePeriod
{
    NIGHT,
    DAWN,
    DAY,
    EVENING
};

static TimePeriod GetTimePeriod(float gameMinutes)
{
    if (gameMinutes >= 5.0f * 60.0f && gameMinutes < 7.0f * 60.0f)
        return TimePeriod::DAWN;

    if (gameMinutes >= 7.0f * 60.0f && gameMinutes < 18.0f * 60.0f)
        return TimePeriod::DAY;

    if (gameMinutes >= 18.0f * 60.0f && gameMinutes < 20.0f * 60.0f)
        return TimePeriod::EVENING;

    return TimePeriod::NIGHT;
}

static const char* GetTimePeriodText(TimePeriod period)
{
    switch (period)
    {
        case TimePeriod::DAWN: return "DAWN";
        case TimePeriod::DAY: return "DAYLIGHT";
        case TimePeriod::EVENING: return "EVENING";
        default: return "NIGHT";
    }
}

static Color GetSkyColor(float gameMinutes)
{
    const Color nightColor = { 5, 8, 18, 255 };
    const Color dawnColor = { 85, 95, 115, 255 };
    const Color dayColor = { 110, 160, 190, 255 };
    const Color eveningColor = { 70, 55, 80, 255 };

    // 00:00 - 05:00 -> night
    if (gameMinutes < 5.0f * 60.0f)
        return nightColor;

    // 05:00 - 07:00 -> sunrise
    if (gameMinutes < 7.0f * 60.0f)
        return LerpColor(nightColor, dayColor,
            (gameMinutes - 5.0f * 60.0f) / (2.0f * 60.0f));

    // 07:00 - 18:00 -> day
    if (gameMinutes < 18.0f * 60.0f)
        return dayColor;

    // 18:00 - 20:00 -> sunset
    if (gameMinutes < 20.0f * 60.0f)
        return LerpColor(dayColor, eveningColor,
            (gameMinutes - 18.0f * 60.0f) / (2.0f * 60.0f));

    // 20:00 - 24:00 -> night
    return nightColor;
}

static float GetNightDarkness(float gameMinutes)
{
    // 0.0 = no extra darkness, 1.0 = maximum overlay.
    // This is only an atmospheric screen overlay; it does not change movement or physics.

    if (gameMinutes < 5.0f * 60.0f)
        return 0.72f;

    if (gameMinutes < 7.0f * 60.0f)
    {
        float t = (gameMinutes - 5.0f * 60.0f) / (2.0f * 60.0f);
        return 0.72f * (1.0f - t);
    }

    if (gameMinutes < 18.0f * 60.0f)
        return 0.0f;

    if (gameMinutes < 20.0f * 60.0f)
    {
        float t = (gameMinutes - 18.0f * 60.0f) / (2.0f * 60.0f);
        return 0.48f * t;
    }

    return 0.72f;
}

static bool UpdateGameClock(
    int& currentDay,
    float& gameMinutes,
    int totalDays,
    float deltaTime,
    float gameMinutesPerSecond)
{
    gameMinutes += gameMinutesPerSecond * deltaTime;

    bool dayChanged = false;

    while (gameMinutes >= 24.0f * 60.0f)
    {
        gameMinutes -= 24.0f * 60.0f;
        currentDay++;
        dayChanged = true;

        if (currentDay > totalDays)
        {
            currentDay = totalDays;
            gameMinutes = 23.0f * 60.0f + 59.0f;
            return dayChanged;
        }
    }

    return dayChanged;
}

            {
                DrawText(
                    "7-DAY CLOCK COMPLETE",
                    SCREEN_WIDTH - 300,
                    25,
                    20,
                    LIGHTGRAY
                );
            }
        }

        if (noteOpen && openedNote >= 0)
        {
            DrawRectangle(
                120, 70,
                SCREEN_WIDTH - 240,
                SCREEN_HEIGHT - 140,
                ColorAlpha(BLACK, 0.92f)
            );

            DrawRectangleLines(
                120, 70,
                SCREEN_WIDTH - 240,
                SCREEN_HEIGHT - 140,
                WHITE
            );

            DrawText(notes[openedNote].title, 160, 110, 32, WHITE);
            DrawText(notes[openedNote].text, 160, 170, 22, LIGHTGRAY);

            DrawText(
                TextFormat("Найдено: день %d", currentDay),
                160, SCREEN_HEIGHT - 150,
                18, GRAY
            );

            DrawText(
                "[E] / [ESC] CLOSE",
                SCREEN_WIDTH - 330,
                SCREEN_HEIGHT - 110,
                20, WHITE
            );
        }

        if (journalOpen)
        {
            DrawRectangle(
                100, 50,
                SCREEN_WIDTH - 200,
                SCREEN_HEIGHT - 100,
                ColorAlpha(BLACK, 0.95f)
            );

            DrawRectangleLines(
                100, 50,
                SCREEN_WIDTH - 200,
                SCREEN_HEIGHT - 100,
                WHITE
            );

            DrawText("ЖУРНАЛ РАССЛЕДОВАНИЯ", 140, 90, 32, WHITE);

            int journalY = 150;
            bool anyFound = false;

            for (int i = 0; i < NOTE_COUNT; i++)
            {
                if (notes[i].found)
                {
                    anyFound = true;

                    DrawText(
                        TextFormat("%d. %s", i + 1, notes[i].title),
                        150, journalY, 22, WHITE
                    );

                    journalY += 35;

                    DrawText(
                        "Запись найдена.",
                        180, journalY, 17, LIGHTGRAY
                    );

                    journalY += 45;
                }
            }

            if (!anyFound)
                DrawText("Пока ничего не найдено.", 150, journalY, 22, LIGHTGRAY);

            DrawText(
                "[TAB] / [ESC] CLOSE",
                SCREEN_WIDTH - 330,
                SCREEN_HEIGHT - 90,
                20, WHITE
            );
        }

        EndDrawing();
    }

    UnloadModel(ground);
    UnloadModel(treeTrunk);
    UnloadModel(treeLeaves);
    UnloadModel(cabinModel);
    UnloadModel(roofModel);
    UnloadModel(tentBody);
    UnloadModel(crateModel);
    UnloadModel(logModel);

    CloseWindow();
    return 0;
}
